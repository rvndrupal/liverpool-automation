npx playwright test — Corre todos los tests
npx playwright test --headed — Corre con browser visible
npx playwright test nombre.spec.js — Corre un archivo específico

npx playwright codegen https://url.com — Graba tus acciones y genera código

# Crear proyecto Playwright
npm init playwright@latest

# Ejecutar todos los tests
npx playwright test

# Ejecutar un archivo específico
npx playwright test login.spec.js

# Ejecutar un test por nombre
npx playwright test -g "Login exitoso"

# Ejecutar en modo UI
npx playwright test --ui

# Ejecutar en modo headed (ver navegador)
npx playwright test --headed

# Debug paso a paso
npx playwright test --debug

# Generar reporte HTML
npx playwright show-report

# Abrir codegen (graba acciones)
npx playwright codegen http://localhost:8000/
npx playwright codegen https://demoqa.com/text-box
